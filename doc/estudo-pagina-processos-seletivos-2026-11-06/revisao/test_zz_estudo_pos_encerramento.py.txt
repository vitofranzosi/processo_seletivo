"""ESTUDO TEMPORÁRIO (não versionar): o que o domínio aceita e recusa depois do encerramento.

S1 = Edital ENCERRADO com Processo ATIVO; S2 = Processo ENCERRADO (Edital encerrado);
S3 = Processo CANCELADO (Edital cancelado); S3b = Processo CANCELADO com Edital ENCERRADO.

O estado final é gravado direto por `update`, como em `test_processo_em_estado_final.py`: o que se
mede é o comando diante do estado, e não o caminho até ele. Cada ato é chamado pelo comando de
aplicação, e o desfecho é registrado em `RESULTADOS` (jsonl no scratchpad) e conferido contra a
matriz esperada.
"""

import json
import os
from decimal import Decimal

import pytest
from django.core import mail
from django.utils import timezone

from processo_seletivo.processos.models import Edital, ProcessoSeletivo
from processo_seletivo.shared.api.problems import DomainError

pytestmark = [pytest.mark.django_db(transaction=True), pytest.mark.integration]

SAIDA = os.environ.get("ESTUDO_SAIDA", "")

SITUACOES = {
    "S1": (Edital.Status.ENCERRADO, ProcessoSeletivo.Status.ATIVO),
    "S2": (Edital.Status.ENCERRADO, ProcessoSeletivo.Status.ENCERRADO),
    "S3": (Edital.Status.CANCELADO, ProcessoSeletivo.Status.CANCELADO),
    "S3b": (Edital.Status.ENCERRADO, ProcessoSeletivo.Status.CANCELADO),
}

FINAL = "O Processo Seletivo está em estado final e não admite alteração de seus Editais."

# A matriz observada na primeira rodada (modo de registro), agora conferida.
_A = ("ACEITO", None)
_F = ("RECUSADO", "invalid_state")
ESPERADO = {}
for _s in SITUACOES:
    _final = _s != "S1"
    for _ato in (
        "e1_apresentar_documento",
        "e2_conferir_documento",
        "f1_requerimento_abrir_rascunho",
        "f2_requerimento_gravar",
        "f3_requerimento_enviar",
        "a_convocar_mais_uma",
    ):
        ESPERADO[(_s, _ato)] = _F if _final else _A
    for _ato in ("b_interpor_recurso", "c1_admitir_recurso", "c2_julgar_recurso"):
        ESPERADO[(_s, _ato)] = _A
    ESPERADO[(_s, "h_exportar_matriculas")] = _A
    ESPERADO[(_s, "g_confirmar_aviso_do_resultado")] = (
        ("RECUSADO", "aviso_processo_em_estado_final") if _final else _A
    )
    ESPERADO[(_s, "d_designar_formalizacao")] = (
        ("RECUSADO", "formalizacao_edital_sem_publicacao") if _s == "S3" else _A
    )


def aplicar(edital, situacao):
    status_do_edital, status_do_processo = SITUACOES[situacao]
    Edital.objects.filter(pk=edital.pk).update(status=status_do_edital)
    ProcessoSeletivo.objects.filter(pk=edital.processo_id).update(status=status_do_processo)


def tentar(situacao, ato, funcao):
    try:
        valor = funcao()
    except DomainError as recusa:
        desfecho = {"r": "RECUSADO", "code": recusa.code, "detail": recusa.detail}
        valor = None
    else:
        desfecho = {"r": "ACEITO"}
    linha = {"situacao": situacao, "ato": ato, **desfecho}
    if SAIDA:
        with open(SAIDA, "a", encoding="utf-8") as arquivo:
            arquivo.write(json.dumps(linha, ensure_ascii=False) + "\n")
    esperado = ESPERADO.get((situacao, ato))
    if esperado is not None:
        assert desfecho["r"] == esperado[0], linha
        if esperado[0] == "RECUSADO":
            assert desfecho["code"] == esperado[1], linha
    return desfecho, valor

def _publicador_na_convocacao():
    from tests.fixtures.publicacao import publish_original
    from tests.fixtures.requerimento import declarar

    def publicar(api_client, manager_headers, process_payload, *, draft=None):
        return publish_original(
            api_client,
            manager_headers,
            process_payload,
            draft=draft,
            antes_de_submeter=declarar("AT_CALL"),
        )

    return publicar


@pytest.mark.parametrize("situacao", list(SITUACOES))
def test_convocacao_documentos_requerimento_designacao(
    situacao, gestor, api_client, manager_headers, process_payload, raiz_de_arquivos
):
    from processo_seletivo.convocacao.application import selectors
    from processo_seletivo.formalizacao.application import designar
    from processo_seletivo.formalizacao.domain import nomes as nomes_da_formalizacao
    from processo_seletivo.inscricoes.models import Inscricao
    from processo_seletivo.requerimentos.application import preencher
    from processo_seletivo.seguranca import acessos as catalogo
    from processo_seletivo.seguranca.domain import Actor
    from tests.fixtures import documentos_da_convocacao as cenarios
    from tests.fixtures.candidato import MARIA
    from tests.fixtures.convocacao import convocar
    from tests.fixtures.corte import MARCO, regra
    from tests.fixtures.edital import PROFILE_ID
    from tests.fixtures.requerimento import campos_de_exemplo
    from tests.seguranca.apoio import pessoa, vincular

    certame = cenarios.montar(
        gestor,
        api_client,
        manager_headers,
        process_payload,
        documentos=cenarios.documentos_do_aluno(),
        prefixo=f"estudo-pos-{situacao}",
        publicar=_publicador_na_convocacao(),
        geral=3,
        cut=regra(surplusCount=1),
    )
    edital, _, inscricoes = certame
    convocada, convocacao = cenarios.convocar_primeira(certame, gestor)
    # Uma apresentação antes do encerramento, para haver o que conferir depois dele.
    cenarios.apresentar(convocada, convocacao, cenarios.HISTORICO)
    rui = pessoa("rui", nome="Rui Registro", email="rui@ifes.edu.br")
    vincular(rui, "cefor", nomes_da_formalizacao.PAPEL)
    gil = Actor("pessoa-gil", "cefor", frozenset({catalogo.GERIR, catalogo.CONSULTAR}))

    aplicar(edital, situacao)
    convocada = Inscricao.objects.select_related("edital").get(pk=convocada.pk)

    # e) apresentar e conferir documento da convocação
    tentar(
        situacao,
        "e1_apresentar_documento",
        lambda: cenarios.apresentar(convocada, convocacao, cenarios.CERTIDAO),
    )
    tentar(
        situacao,
        "e2_conferir_documento",
        lambda: cenarios.conferir(gestor, convocacao, cenarios.HISTORICO, "CONFORME", chave="c"),
    )

    # f) Requerimento de Matrícula na convocação
    campos = campos_de_exemplo()
    tentar(
        situacao,
        "f1_requerimento_abrir_rascunho",
        lambda: preencher.abrir_rascunho(inscricao=convocada),
    )
    tentar(
        situacao,
        "f2_requerimento_gravar",
        lambda: preencher.gravar(inscricao=convocada, dados=campos, expected_revision=None),
    )

    def enviar():
        versao = preencher._conteudo(convocada)
        return preencher.enviar(
            identidade=MARIA,
            inscricao=convocada,
            versao_exibida_id=versao.id,
            declaracao_exibida=preencher.declaracao_publicada(versao.content),
            aceite=True,
        )

    tentar(situacao, "f3_requerimento_enviar", enviar)

    # d) designar formalização
    tentar(
        situacao,
        "d_designar_formalizacao",
        lambda: designar.designar(
            gil,
            edital_id=edital.id,
            pessoa=rui,
            perfis=[],
            motivo="Estudo pós-encerramento",
            now=timezone.now(),
        ),
    )

    # a) convocar mais uma pessoa: a próxima da fila que não é a já convocada
    fila = selectors.contexto_do_recorte(
        edital=edital, perfil_id=PROFILE_ID, marco_id=MARCO, lista_id=None
    )["fila"]
    proxima = next(i for i in fila if str(i) != str(convocada.id))
    _, declarado = tentar(
        situacao,
        "a_convocar_mais_uma",
        lambda: convocar(edital, gestor, proxima, idempotency_key="estudo-convocar-2"),
    )
    if declarado is not None and SAIDA:
        with open(SAIDA, "a", encoding="utf-8") as arquivo:
            linha = {"situacao": situacao, "ato": "a_especie", "r": str(declarado.get("especie"))}
            arquivo.write(json.dumps(linha, ensure_ascii=False) + "\n")
    assert inscricoes  # o cenário tem fila

@pytest.mark.parametrize("situacao", list(SITUACOES))
def test_recursos(situacao, gestor, api_client, manager_headers, process_payload):
    from processo_seletivo.portal.identidade import IdentidadeDoCandidato
    from processo_seletivo.recursos.application.admitir import admitir
    from processo_seletivo.recursos.application.interpor import interpor
    from processo_seletivo.recursos.application.julgar import julgar
    from processo_seletivo.recursos.models import DecisaoRecurso
    from processo_seletivo.resultados.models import ResultadoEtapa
    from tests.fixtures.recursos_us4 import assinatura_de, cenario_julgavel, julgador

    peca = cenario_julgavel(
        gestor,
        api_client,
        manager_headers,
        process_payload,
        seed=910,
        codigo="0910",
        admitir_a_peca=False,
    )
    edital = peca["inscricao"].edital
    outra = peca["cenario"]["inscricoes"][1]
    etapa = peca["cenario"]["etapa"]
    resultado_da_outra = ResultadoEtapa.vigentes.get(inscricao=outra, etapa_id=etapa)

    aplicar(edital, situacao)

    def interpor_de_novo():
        from processo_seletivo.inscricoes.models import Inscricao

        relida = Inscricao.objects.select_related("edital").get(pk=outra.pk)
        return interpor(
            identidade=IdentidadeDoCandidato(
                relida.identity_subject, relida.nome, relida.cpf_normalizado, "c@ex.br"
            ),
            inscricao=relida,
            resultado=resultado_da_outra,
            fundamentacao="Recurso interposto depois do encerramento.",
            assinatura_do_objeto=str(resultado_da_outra.pk),
            idempotency_key="estudo-interpor-depois",
        )

    tentar(situacao, "b_interpor_recurso", interpor_de_novo)
    tentar(
        situacao,
        "c1_admitir_recurso",
        lambda: admitir(
            actor=julgador(),
            recurso_id=peca["recurso"].id,
            admitido=True,
            motivo="Tempestivo e regularmente instruído.",
            assinatura_do_estado=assinatura_de(peca["recurso"]),
            idempotency_key="estudo-admitir",
        ),
    )
    _, julgado = tentar(
        situacao,
        "c2_julgar_recurso",
        lambda: julgar(
            actor=julgador(),
            recurso_id=peca["recurso"].id,
            especie=DecisaoRecurso.Especie.CORRECAO_FIXADA,
            motivacao="O parecer não enfrentou o documento juntado na inscrição.",
            etapa_id=etapa,
            pontuacao=Decimal("82.0000"),
            assinatura_do_resultado=str(peca["superado"].id),
            idempotency_key="estudo-julgar",
        ),
    )
    if julgado is not None and SAIDA:
        _, sucessor = julgado
        with open(SAIDA, "a", encoding="utf-8") as arquivo:
            linha = {
                "situacao": situacao,
                "ato": "c2_sucessor_do_resultado",
                "r": f"sucessor={'sim' if sucessor is not None else 'não'}",
            }
            arquivo.write(json.dumps(linha, ensure_ascii=False) + "\n")

@pytest.mark.parametrize("situacao", list(SITUACOES))
def test_aviso_confirmado_antes_e_despachado_depois(
    situacao, gestor, api_client, manager_headers, process_payload
):
    """O aviso confirmado com o Processo ativo ainda sai depois do encerramento?"""
    from processo_seletivo.avisos.application.despacho import despachar
    from tests.fixtures.avisos import avisar_resultado
    from tests.fixtures.divulgacao import montar_ato_publicavel, publicar_o_ato

    cenario = montar_ato_publicavel(
        gestor, api_client, manager_headers, process_payload, pontuacoes=("90.0000", "70.0000")
    )
    publicacao = publicar_o_ato(cenario)
    mail.outbox.clear()
    avisar_resultado(cenario["edital"], publicacao.marco_id)
    aplicar(cenario["edital"], situacao)

    resumo = despachar()
    linha = {
        "situacao": situacao,
        "ato": "g2_despachar_aviso_confirmado_antes",
        "r": f"tentadas={resumo.tentadas} resultados={resumo.resultados} "
        f"mensagens={len(mail.outbox)}",
    }
    if SAIDA:
        with open(SAIDA, "a", encoding="utf-8") as arquivo:
            arquivo.write(json.dumps(linha, ensure_ascii=False) + "\n")
    assert len(mail.outbox) == 2, linha

@pytest.mark.parametrize("situacao", list(SITUACOES))
def test_exportar_matriculas(
    situacao, gestor, api_client, manager_headers, process_payload, raiz_de_arquivos
):
    from processo_seletivo.matriculas.application.exportar import compor, gerar
    from processo_seletivo.matriculas.application.populacao import opcoes
    from processo_seletivo.matriculas.domain import nomes
    from tests.conftest import ator_institucional
    from tests.fixtures.matriculas import montar_cenario_da_exportacao

    edital, _ = montar_cenario_da_exportacao(gestor, api_client, manager_headers, process_payload)
    quem_exporta = ator_institucional("exportadora", nomes.EXPORTAR)
    aplicar(edital, situacao)
    edital = Edital.objects.get(pk=edital.pk)

    def exportar():
        escolhida = opcoes(edital)[0]
        argumentos = {
            "ator": quem_exporta,
            "edital": edital,
            "especie": escolhida.especie,
            "referencia": escolhida.referencia,
        }
        return gerar(**argumentos, confirmacao_do_resumo=compor(**argumentos).assinatura)

    tentar(situacao, "h_exportar_matriculas", exportar)
